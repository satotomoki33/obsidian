```
この変更の理由は、STATE_COLLECTING というROS向けの公開状態と、MfCollectSequence の内部FSM状態が混ざっていて読みにくかったためです。

以前は set_ros_state(uint8_t) に MfCollectState::STATE_COLLECTING を渡すと、内部では InitializedState に遷移していました。名前だけ見ると「回収中にする」ように見えるのに、実際は「初期化完了後、次の回収目標を開始できる状態」になっていて、意図が分かりづらい実装でした。

そこで、外向きのROS state enumを内部FSMへ逆変換するのをやめて、set_planning()、set_following()、set_failed() のように内部状態の意味がそのまま分かるメソッドへ分けました。機構動作については start_initializing() と start_collecting() だけから開始する形にしたので、STATE_COLLECTING を外から直接 set する曖昧な経路がなくなりました。

結果として、呼び出し側を読んだときに「今どの内部状態へ遷移させたいのか」が明確になり、STATE_COLLECTING が InitializedState / CollectingState / TargetCompletedState など複数の内部状態を代表していることによる混乱を避けられます。ROSにpublishする状態と、内部FSMを進める操作の責務を分けた、というのが一番大きい理由です
```

```
#include <chrono>

#include <cmath>

#include <limits>

#include <optional>

#include <rtcl/core/log.hpp>

#include <rtcl/payload/motor.hpp>

#include <string>

#include <utility>

  

#include "dog3_rtcl/payload/mechanism.hpp"

#include "dog3_rtcl/util/calibration/controller.hpp"

#include "dog3_rtcl/util/motor_command_config_validation.hpp"

#include "dog3_rtcl/util/motor_config_conversion.hpp"

  

namespace dog3_rtcl {

namespace {

constexpr auto kFeedbackTimeout = std::chrono::milliseconds(500);

  

auto validate_lift_parameter(const StaffLiftControlBlock::Parameter& parameter) -> asl::SetParameterResult {

if (!std::isfinite(parameter.meter_per_rev) || parameter.meter_per_rev <= 0.0f) {

return asl::SetParameterResult::error("meter_per_rev は正の有限値である必要があります");

}

if (!std::isfinite(parameter.min_position_m) || !std::isfinite(parameter.max_position_m)) {

return asl::SetParameterResult::error("min_position_m/max_position_m は有限値である必要があります");

}

if (parameter.min_position_m >= parameter.max_position_m) {

return asl::SetParameterResult::error("min_position_m は max_position_m より小さい必要があります");

}

if (const auto result = validate_stopper_calibration_parameter(parameter.calibration); result.is_error()) {

return result;

}

if (const auto result = validate_position_config(parameter.position_config); result.is_error()) {

return result;

}

if (const auto result = validate_calibration_velocity_config(parameter.calibration_velocity_config);

result.is_error()) {

return result;

}

return asl::SetParameterResult::ok();

}

  

} // namespace

  

class StaffLiftControlBlock::Impl {

public:

Impl(rtcl::BlockBuildContext& ctx, std::string name, ExecutorSpec executor, MotorRequesterSpec lifter_motor_requester,

rtcl::Read<rtcl_blocks::EmergencyStopState> emergency_stop_reader)

: name_(std::move(name))

, executor_(ctx.act().bind(executor))

, requester_(ctx.act().bind(lifter_motor_requester))

, emergency_stop_reader_(ctx.act().bind(emergency_stop_reader)) {}

  

auto run_act(const rtcl::TickContext& ctx) -> void {

const auto* emergency_stop = emergency_stop_reader_.read(ctx);

const bool emergency_active = emergency_stop ? emergency_stop->active : true;

const auto* feedback = requester_.latest_feedback_within(ctx, kFeedbackTimeout);

const auto position_rad = read_position_rad(feedback);

  

if (emergency_active) {

target_position_m_.reset();

stop_hold_active_ = false;

calibration_.reset();

set_zero_torque(ctx);

publish_feedback(ctx, emergency_active, position_rad);

return;

}

  

if (const auto* command = executor_.read_command(ctx); command != nullptr) {

handle_command(*command, position_rad);

}

  

update_motor_command(ctx, feedback, position_rad);

publish_feedback(ctx, emergency_active, position_rad);

}

  

auto handle_command(const StaffLiftCommand& command, const std::optional<float>& position_rad) -> void {

switch (command.type) {

case MechanismCommand::Target:

if (!std::isfinite(command.lifter_position_m)) {

rtcl::log_throttle("staff_lift_nan_command", std::chrono::seconds(1))

.info("StaffLiftControlBlock: 目標位置がNaNのコマンドを受信しました。コマンドを無視します。");

return;

}

stop_hold_active_ = false;

set_target_position(command.lifter_position_m);

if (!calibration_.calibrated()) {

start_stopper_calibration();

}

return;

  

case MechanismCommand::Calibrate:

stop_hold_active_ = false;

target_position_m_ = parameter_.min_position_m;

start_stopper_calibration();

return;

  

case MechanismCommand::ResetCalibration:

stop_hold_active_ = false;

target_position_m_.reset();

calibration_.reset();

return;

  

case MechanismCommand::Stop:

if (stop_hold_active_) {

return;

}

if (calibration_.calibrating()) {

target_position_m_.reset();

calibration_.reset();

} else {

hold_current_position(position_rad);

}

stop_hold_active_ = true;

return;

}

}

  

auto set_target_position(float position_m) -> void {

if (position_m < parameter_.min_position_m || position_m > parameter_.max_position_m) {

rtcl::log_throttle("staff_lift_position_out_of_bounds", std::chrono::seconds(1))

.warn("StaffLiftControlBlock: 目標位置が範囲外です: {:.6f} m (許容範囲: [{:.2f}, {:.2f}] m)", position_m,

parameter_.min_position_m, parameter_.max_position_m);

}

target_position_m_ = std::clamp(position_m, parameter_.min_position_m, parameter_.max_position_m);

}

  

auto hold_current_position(const std::optional<float>& position_rad) -> void {

if (!position_rad.has_value() || !calibration_.calibrated()) {

target_position_m_.reset();

return;

}

const auto current_position_m = current_position(*position_rad);

if (!current_position_m.has_value()) {

target_position_m_.reset();

return;

}

set_target_position(*current_position_m);

}

  

auto start_stopper_calibration() -> void {

if (calibration_.stopper_calibration_active()) {

return;

}

const auto res =

calibration_.request_stopper_calibration(parameter_.calibration, parameter_.reverse_direction, false);

if (res == calibration::CalibrationStartResult::Failed) {

rtcl::log_throttle("StaffLiftControlBlock", std::chrono::seconds(1))

.error("スタッフリフトの下端ストッパーキャリブレーション開始に失敗しました");

target_position_m_.reset();

calibration_.fail();

}

}

  

auto update_motor_command(const rtcl::TickContext& ctx, const rtcl::MotorFeedback* feedback,

const std::optional<float>& position_rad) -> void {

if (calibration_.stopper_calibration_active()) {

update_stopper_calibration(ctx, feedback, position_rad);

return;

}

  

if (!target_position_m_.has_value()) {

set_zero_torque(ctx);

return;

}

  

if (!calibration_.calibrated()) {

start_stopper_calibration();

if (calibration_.stopper_calibration_active()) {

update_stopper_calibration(ctx, feedback, position_rad);

} else {

set_zero_torque(ctx);

}

return;

}

  

if (!position_rad.has_value()) {

target_position_m_.reset();

set_zero_torque(ctx);

return;

}

  

const auto target_rad = target_position_rad(*target_position_m_);

if (!target_rad.has_value()) {

target_position_m_.reset();

set_zero_torque(ctx);

return;

}

  

requester_.set_command(ctx, rtcl::MotorPositionCommand{

.target_position = *target_rad,

.ff_torque = 0.0f,

.config = parameter_.position_config,

});

}

  

auto update_stopper_calibration(const rtcl::TickContext& ctx, const rtcl::MotorFeedback* feedback,

const std::optional<float>& position_rad) -> void {

const auto velocity_radps = read_velocity_radps(feedback);

if (!position_rad.has_value() || !velocity_radps.has_value()) {

target_position_m_.reset();

calibration_.fail();

set_zero_torque(ctx);

return;

}

  

calibration_.update_stopper_calibration(calibration::StopperCalibration::Input{

.actuator_position = *position_rad,

.actuator_velocity = *velocity_radps,

});

  

if (!calibration_.stopper_calibration_active()) {

set_zero_torque(ctx);

if (!calibration_.calibrated()) {

target_position_m_.reset();

}

return;

}

  

requester_.set_command(ctx, rtcl::MotorVelocityCommand{

.target_velocity = calibration_.target_velocity(),

.ff_torque = 0.0f,

.config = parameter_.calibration_velocity_config,

});

}

  

auto set_zero_torque(const rtcl::TickContext& ctx) -> void {

requester_.set_command(ctx, rtcl::MotorTorqueCommand{.target_torque = 0.0f});

}

  

[[nodiscard]] auto current_position(float position_rad) const -> std::optional<float> {

const auto artifact = calibration_.artifact();

if (!artifact.has_value()) {

return std::nullopt;

}

return parameter_.min_position_m + rad_to_meter(position_rad - artifact->actuator_position_at_command_zero,

parameter_.meter_per_rev, parameter_.reverse_direction);

}

  

[[nodiscard]] auto target_position_rad(float position_m) const -> std::optional<float> {

const auto artifact = calibration_.artifact();

if (!artifact.has_value()) {

return std::nullopt;

}

const float clamped_position_m = std::clamp(position_m, parameter_.min_position_m, parameter_.max_position_m);

const float distance_from_origin_m = clamped_position_m - parameter_.min_position_m;

return artifact->actuator_position_at_command_zero +

meter_to_rad(distance_from_origin_m, parameter_.meter_per_rev, parameter_.reverse_direction);

}

  

auto publish_feedback(const rtcl::TickContext& ctx, bool emergency_active, const std::optional<float>& position_rad)

-> void {

float position_m = std::numeric_limits<float>::quiet_NaN();

if (position_rad.has_value() && calibration_.calibrated()) {

if (const auto current_position_m = current_position(*position_rad); current_position_m.has_value()) {

position_m = *current_position_m;

}

}

  

executor_.set_feedback(ctx, StaffLiftFeedback{.state = state(emergency_active, position_rad.has_value()),

.lifter_position_m = position_m});

}

  

[[nodiscard]] auto state(bool emergency_active, bool feedback_available) const -> MechanismState {

if (emergency_active) {

return MechanismState::Init;

}

if (calibration_.failed()) {

return MechanismState::Error;

}

if (calibration_.calibrating()) {

return MechanismState::Calibrating;

}

if (calibration_.calibrated() && !feedback_available) {

return MechanismState::Error;

}

if (calibration_.calibrated() && feedback_available) {

return MechanismState::Active;

}

return MechanismState::Init;

}

  

std::string name_;

rtcl::Executor<StaffLiftCommand, StaffLiftFeedback> executor_{};

rtcl::Requester<rtcl::MotorCommand, rtcl::MotorFeedback> requester_{};

rtcl::Reader<rtcl_blocks::EmergencyStopState> emergency_stop_reader_{};

Parameter parameter_{};

std::optional<float> target_position_m_{};

bool stop_hold_active_{false};

calibration::CalibrationController calibration_{calibration::CalibrationReusePolicy::AlwaysRecalibrate};

};

  

StaffLiftControlBlock::StaffLiftControlBlock(rtcl::BlockBuildContext& ctx, std::string name, ExecutorSpec executor,

MotorRequesterSpec lifter_motor_requester,

rtcl::Read<rtcl_blocks::EmergencyStopState> emergency_stop_reader)

: impl_(std::make_unique<Impl>(ctx, std::move(name), executor, lifter_motor_requester, emergency_stop_reader)) {}

  

StaffLiftControlBlock::~StaffLiftControlBlock() = default;

StaffLiftControlBlock::StaffLiftControlBlock(StaffLiftControlBlock&&) noexcept = default;

auto StaffLiftControlBlock::operator=(StaffLiftControlBlock&&) noexcept -> StaffLiftControlBlock& = default;

  

auto StaffLiftControlBlock::name() const -> std::string_view {

return impl_->name_;

}

  

auto StaffLiftControlBlock::run_act(const rtcl::TickContext& ctx) -> void {

return impl_->run_act(ctx);

}

  

auto StaffLiftControlBlock::get_parameter() const -> Parameter {

return impl_->parameter_;

}

  

auto StaffLiftControlBlock::set_parameter(const Parameter& parameter) -> asl::SetParameterResult {

if (const auto result = validate_lift_parameter(parameter); result.is_error()) {

return result;

}

impl_->parameter_ = parameter;

return asl::SetParameterResult::ok();

}

  

} // namespace dog3_rtcl
```
